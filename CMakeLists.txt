cmake_minimum_required(VERSION 3.21)
set(FRETWAVE_VERSION "0.1.0" CACHE STRING "Fretwave release version")
project(Fretwave VERSION ${FRETWAVE_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickControls2 Multimedia)
find_package(KF6Kirigami REQUIRED)
qt_standard_project_setup(REQUIRES 6.5)
if(Qt6_VERSION VERSION_GREATER_EQUAL 6.8)
  qt_policy(SET QTP0004 NEW)
endif()
qt_add_executable(fretwave src/main.cpp src/chords.cpp src/chords.h src/synth.cpp src/synth.h src/audioengine.cpp src/audioengine.h)
qt_add_resources(fretwave "branding" PREFIX "/" FILES icons/fretwave.svg icons/fretwave-256.png)
qt_add_qml_module(fretwave URI Fretwave VERSION 1.0 QML_FILES qml/Main.qml qml/ChordDiagram.qml qml/NotesWheel.qml)
qt_add_resources(fretwave "fonts" PREFIX "/" FILES
  fonts/Oswald.ttf fonts/Oswald-OFL.txt
  fonts/Lato-Regular.ttf fonts/Lato-Bold.ttf fonts/Lato-OFL.txt
  fonts/NotoSans-Regular.ttf fonts/OFL.txt)
target_link_libraries(fretwave PRIVATE Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia)
include(GNUInstallDirs)
install(TARGETS fretwave RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES io.github.codella.fretwave.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES icons/fretwave.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME io.github.codella.fretwave.svg)
foreach(size 16 32 48 64 128 256)
  install(FILES icons/fretwave-${size}.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/${size}x${size}/apps RENAME io.github.codella.fretwave.png)
endforeach()
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/fretwave)
install(FILES fonts/OFL.txt fonts/Oswald-OFL.txt fonts/Lato-OFL.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/fretwave/fonts)
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(synth-tests tests/synth_test.cpp src/synth.cpp src/synth.h)
  target_include_directories(synth-tests PRIVATE src)
  target_link_libraries(synth-tests PRIVATE Qt6::Core)
  add_test(NAME audio-synthesis COMMAND synth-tests)
  qt_add_executable(chord-tests tests/chords_test.cpp src/chords.cpp src/chords.h)
  target_include_directories(chord-tests PRIVATE src)
  target_link_libraries(chord-tests PRIVATE Qt6::Core)
  add_test(NAME chord-integrity COMMAND chord-tests)
  add_test(NAME ui-workflow COMMAND fretwave --verify-ui)
  set_tests_properties(ui-workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
